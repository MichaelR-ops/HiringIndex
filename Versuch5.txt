import re
import requests

BASE = "https://jobs.volkswagen-group.com/search"

for offset in [0, 25, 50, 75]:
    params = {
        "q": "",
        "searchby": "location",
        "startrow": offset,
    }
    r = requests.get(BASE, params=params, headers={"User-Agent": "Mozilla/5.0"})
    

    idx = r.text.find("Results")
    print(r.text[idx-50:idx+300])
